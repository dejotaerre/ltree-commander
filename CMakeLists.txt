cmake_minimum_required(VERSION 3.22)
project(ltree_commander VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Widgets Concurrent Network PrintSupport)
find_package(qtermwidget6 REQUIRED)
find_package(LibArchive REQUIRED)

add_library(ltree_core src/core/helpdocument.cpp src/core/session.cpp src/core/filespec.cpp src/core/sort.cpp src/fs/scanner.cpp src/fs/search.cpp src/fs/mounts.cpp src/fs/viewdocument.cpp src/fs/hexedit.cpp src/fs/archive.cpp src/fs/filecompare.cpp src/fs/makedirectory.cpp src/fs/deletedirectory.cpp src/fs/prune.cpp src/fs/graft.cpp src/fs/copyfiles.cpp src/fs/metadata.cpp src/fs/stamp.cpp src/fs/systemstats.cpp src/fs/rename.cpp src/fs/directorycompare.cpp src/fs/symlink.cpp)
target_include_directories(ltree_core PUBLIC src)
target_link_libraries(ltree_core PUBLIC Qt6::Core LibArchive::LibArchive)
target_compile_options(ltree_core PRIVATE -Wall -Wextra -Wpedantic)

add_library(ltree_ui src/ui/windowgeometry.cpp src/ui/helpview.cpp src/ui/treewindow.cpp src/ui/treewindowarchive.cpp src/ui/treewindowdir.cpp src/ui/treewindowprune.cpp src/ui/treewindowgraft.cpp src/ui/treewindowcopy.cpp src/ui/treewindowmetadata.cpp src/ui/treewindowstamp.cpp src/ui/treewindowhistory.cpp src/ui/treewindowstats.cpp src/ui/treewindowrename.cpp src/ui/treewindowdirectorycompare.cpp src/ui/consolefont.cpp src/ui/fileviewer.cpp src/ui/fileviewercommands.cpp src/ui/archivewindow.cpp src/ui/comparewindow.cpp src/ui/executewindow.cpp)
target_link_libraries(ltree_ui PUBLIC ltree_core Qt6::Widgets Qt6::Concurrent Qt6::PrintSupport qtermwidget6)
target_compile_options(ltree_ui PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(ltree_ui PUBLIC LTREE_VERSION="${PROJECT_VERSION}-alpha.1")

find_program(LTREE_FONT_PYTHON NAMES python3 REQUIRED)
execute_process(COMMAND "${LTREE_FONT_PYTHON}" -c "import fontTools"
    RESULT_VARIABLE FONTTOOLS_STATUS OUTPUT_QUIET ERROR_QUIET)
if(NOT FONTTOOLS_STATUS EQUAL 0)
    message(FATAL_ERROR "Python fontTools is required to generate the console font")
endif()
set(LTREE_PSF_FONT "/usr/share/kbd/consolefonts/default8x16.psfu.gz" CACHE FILEPATH "Installed PSF2 8x16 font")
if(NOT EXISTS "${LTREE_PSF_FONT}")
    message(FATAL_ERROR "Install the kbd console font or set LTREE_PSF_FONT to a PSF2 8x16 font")
endif()
target_compile_definitions(ltree_ui PRIVATE LTREE_PSF_FONT_PATH="${LTREE_PSF_FONT}")
set(LTREE_CONSOLE_TTF "${CMAKE_CURRENT_BINARY_DIR}/resources/console8x16.ttf")
add_custom_command(OUTPUT "${LTREE_CONSOLE_TTF}"
    COMMAND "${LTREE_FONT_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_console_font.py"
        "${LTREE_PSF_FONT}" "${LTREE_CONSOLE_TTF}"
    DEPENDS tools/build_console_font.py "${LTREE_PSF_FONT}" VERBATIM)
set_source_files_properties("${LTREE_CONSOLE_TTF}" PROPERTIES GENERATED TRUE)
qt_add_resources(ltree_ui execute_font PREFIX "/" BASE "${CMAKE_CURRENT_BINARY_DIR}" FILES "${LTREE_CONSOLE_TTF}")

option(LTREE_BUILD_TERMINAL "Build the experimental ncurses terminal interface" ON)
if(LTREE_BUILD_TERMINAL)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(NCURSESW REQUIRED IMPORTED_TARGET ncursesw)
    add_library(ltree_terminal src/terminal/terminalui.cpp)
    target_link_libraries(ltree_terminal PUBLIC ltree_core Qt6::Concurrent PkgConfig::NCURSESW)
    target_compile_options(ltree_terminal PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(ltc src/main.cpp)
qt_add_resources(ltc desktop_icon PREFIX "/icons" BASE "resources/icons" FILES "resources/icons/ltreec.png")
target_link_libraries(ltc PRIVATE ltree_ui Qt6::Network)
if(LTREE_BUILD_TERMINAL)
    target_link_libraries(ltc PRIVATE ltree_terminal)
    target_compile_definitions(ltc PRIVATE LTREE_HAS_TERMINAL)
endif()

    add_test(NAME ltree-help-tests COMMAND ltree-help-tests)

    add_dependencies(ltree-tests ltc)
