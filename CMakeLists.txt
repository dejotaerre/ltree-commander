cmake_minimum_required(VERSION 3.22)
project(ltree_commander VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LTREE_BUILD_GUI "Build the graphical interface" ON)
option(LTREE_BUILD_TERMINAL "Build the ncurses terminal interface" ON)
if(NOT LTREE_BUILD_GUI AND NOT LTREE_BUILD_TERMINAL)
    message(FATAL_ERROR "Enable LTREE_BUILD_GUI or LTREE_BUILD_TERMINAL")
endif()
if(LTREE_BUILD_GUI)
    set(CMAKE_AUTOMOC ON)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Core Concurrent Widgets Network PrintSupport)
    find_package(qtermwidget6 REQUIRED)
endif()
find_package(LibArchive REQUIRED)
if(NOT LTREE_BUILD_GUI)
    find_package(Threads REQUIRED)
    find_package(ICU REQUIRED COMPONENTS uc)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PCRE2 REQUIRED IMPORTED_TARGET libpcre2-16)
    find_package(nlohmann_json 3.2 REQUIRED)
    add_library(ltree_platform src/platform/portable.cpp)
    target_include_directories(ltree_platform PUBLIC src)
    target_link_libraries(ltree_platform PUBLIC ICU::uc PkgConfig::PCRE2 nlohmann_json::nlohmann_json Threads::Threads)
    target_compile_options(ltree_platform PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(ltree_platform PUBLIC LTREE_VERSION=\"${PROJECT_VERSION}-alpha.1\")
endif()

add_library(ltree_core src/core/helpdocument.cpp src/core/session.cpp src/core/filespec.cpp src/core/sort.cpp src/fs/scanner.cpp src/fs/fileattributes.cpp src/fs/search.cpp src/fs/mounts.cpp src/fs/viewdocument.cpp src/fs/hexedit.cpp src/fs/archive.cpp src/fs/filecompare.cpp src/fs/makedirectory.cpp src/fs/deletedirectory.cpp src/fs/prune.cpp src/fs/graft.cpp src/fs/copyfiles.cpp src/fs/metadata.cpp src/fs/stamp.cpp src/fs/systemstats.cpp src/fs/rename.cpp src/fs/directorycompare.cpp src/fs/symlink.cpp)
target_include_directories(ltree_core PUBLIC src)
target_link_libraries(ltree_core PUBLIC LibArchive::LibArchive)
if(LTREE_BUILD_GUI)
    target_link_libraries(ltree_core PUBLIC Qt6::Core Qt6::Concurrent)
    target_compile_definitions(ltree_core PUBLIC LTREE_USE_QT)
else()
    target_link_libraries(ltree_core PUBLIC ltree_platform)
endif()
target_compile_options(ltree_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(ltree_core PUBLIC LTREE_VERSION=\"${PROJECT_VERSION}-alpha.1\")

if(LTREE_BUILD_GUI)
    add_library(ltree_ui src/ui/windowgeometry.cpp src/ui/helpview.cpp src/ui/treewindow.cpp src/ui/treewindowarchive.cpp src/ui/treewindowdir.cpp src/ui/treewindowprune.cpp src/ui/treewindowgraft.cpp src/ui/treewindowcopy.cpp src/ui/treewindowmetadata.cpp src/ui/treewindowstamp.cpp src/ui/treewindowhistory.cpp src/ui/treewindowstats.cpp src/ui/treewindowrename.cpp src/ui/treewindowdirectorycompare.cpp src/ui/consolefont.cpp src/ui/fileviewer.cpp src/ui/fileviewercommands.cpp src/ui/archivewindow.cpp src/ui/comparewindow.cpp src/ui/executewindow.cpp)
    target_link_libraries(ltree_ui PUBLIC ltree_core Qt6::Widgets Qt6::Concurrent Qt6::PrintSupport qtermwidget6)
    target_compile_options(ltree_ui PRIVATE -Wall -Wextra -Wpedantic)

    find_program(LTREE_FONT_PYTHON NAMES python3 REQUIRED)
    execute_process(COMMAND "${LTREE_FONT_PYTHON}" -c "import fontTools"
        RESULT_VARIABLE FONTTOOLS_STATUS OUTPUT_QUIET ERROR_QUIET)
    if(NOT FONTTOOLS_STATUS EQUAL 0)
        message(FATAL_ERROR "Python fontTools is required to generate the console font")
    endif()
    set(LTREE_PSF_FONT "/usr/share/kbd/consolefonts/default8x16.psfu.gz" CACHE FILEPATH "Installed PSF2 8x16 font")
    if(NOT EXISTS "${LTREE_PSF_FONT}")
        message(FATAL_ERROR "Install the kbd console font or set LTREE_PSF_FONT to a PSF2 8x16 font")
    endif()
    target_compile_definitions(ltree_ui PRIVATE LTREE_PSF_FONT_PATH="${LTREE_PSF_FONT}")
    set(LTREE_CONSOLE_TTF "${CMAKE_CURRENT_BINARY_DIR}/resources/console8x16.ttf")
    add_custom_command(OUTPUT "${LTREE_CONSOLE_TTF}"
        COMMAND "${LTREE_FONT_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_console_font.py"
            "${LTREE_PSF_FONT}" "${LTREE_CONSOLE_TTF}"
        DEPENDS tools/build_console_font.py "${LTREE_PSF_FONT}" VERBATIM)
    set_source_files_properties("${LTREE_CONSOLE_TTF}" PROPERTIES GENERATED TRUE)
    qt_add_resources(ltree_ui execute_font PREFIX "/" BASE "${CMAKE_CURRENT_BINARY_DIR}" FILES "${LTREE_CONSOLE_TTF}")

endif()

if(LTREE_BUILD_TERMINAL)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(NCURSESW REQUIRED IMPORTED_TARGET ncursesw)
    add_library(ltree_terminal src/terminal/terminalui.cpp src/terminal/terminalcommands.inc
        src/terminal/terminalhistory.inc
        src/terminal/terminalfileops.inc
        src/terminal/terminalarchive.inc
        src/terminal/terminalcompare.inc
        src/terminal/terminalnavigation.inc
        src/terminal/terminalviewer.inc
        src/terminal/terminaldispatch.inc
        src/terminal/editor.cpp src/terminal/keyboard.cpp)
    target_link_libraries(ltree_terminal PUBLIC ltree_core PkgConfig::NCURSESW)
    target_compile_options(ltree_terminal PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(LTREE_BUILD_GUI)
    add_executable(ltc src/main.cpp)
else()
    add_executable(ltc src/terminal/main.cpp)
endif()
target_link_libraries(ltc PRIVATE ltree_core)
if(LTREE_BUILD_GUI)
    qt_add_resources(ltc desktop_icon PREFIX "/icons" BASE "resources/icons" FILES "resources/icons/ltreec.png")
    target_link_libraries(ltc PRIVATE ltree_ui Qt6::Network)
    target_compile_definitions(ltc PRIVATE LTREE_HAS_GUI)
endif()
if(LTREE_BUILD_TERMINAL)
    target_link_libraries(ltc PRIVATE ltree_terminal)
    target_compile_definitions(ltc PRIVATE LTREE_HAS_TERMINAL)
endif()

include(CTest)

    if(LTREE_BUILD_TERMINAL)

            target_link_libraries(ltree-terminal-editor-tests PRIVATE ltree_terminal Qt6::Test)
            add_test(NAME ltree-terminal-editor-tests COMMAND ltree-terminal-editor-tests)
        endif()
        foreach(suite terminal_pty terminal_extended_pty terminal_operations_pty terminal_kitty_pty)
            add_test(NAME ltree-${suite} COMMAND ${CMAKE_COMMAND} -E env
                "LTC_TEST_BINARY=$<TARGET_FILE:ltc>" "LTREE_PTY_WORK=" "LTC_TEST_HAS_GUI=$<BOOL:${LTREE_BUILD_GUI}>"

    if(LTREE_BUILD_GUI)
        add_executable(ltree-geometry-tests tests/test_geometry.cpp)

        add_test(NAME ltree-geometry-tests COMMAND ltree-geometry-tests)
        set_tests_properties(ltree-geometry-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        add_executable(ltree-help-tests tests/test_help.cpp)
        target_link_libraries(ltree-help-tests PRIVATE ltree_ui Qt6::Test)
        add_test(NAME ltree-help-tests COMMAND ltree-help-tests)
        set_tests_properties(ltree-help-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        add_executable(ltree-tests tests/test_ltree.cpp)
        target_link_libraries(ltree-tests PRIVATE ltree_ui Qt6::Test)
        add_dependencies(ltree-tests ltc)
        add_executable(ltree-view-archive-tests tests/test_view_archive.cpp)
        target_link_libraries(ltree-view-archive-tests PRIVATE ltree_ui Qt6::Test)
        add_test(NAME ltree-view-archive-tests COMMAND ltree-view-archive-tests)
        set_tests_properties(ltree-view-archive-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        add_executable(ltree-stamp-tests tests/test_stamp.cpp)
        target_link_libraries(ltree-stamp-tests PRIVATE ltree_ui Qt6::Test)
        add_test(NAME ltree-stamp-tests COMMAND ltree-stamp-tests)
        set_tests_properties(ltree-stamp-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        add_test(NAME ltree-tests COMMAND ltree-tests)
        set_tests_properties(ltree-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
